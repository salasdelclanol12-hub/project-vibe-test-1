// js/bridge.js
// Все вызовы Notibot Bridge — только отсюда.

let _state = { user: null, app: null, colors: null };
const _listeners = [];

/**
 * Инициализация Bridge. Вызывается один раз из app.js.
 * @param {Function} onReady — коллбэк { user, app, colors }
 */
export function initBridge(onReady) {
  let resolved = false;
  const handleReady = (state) => {
    if (resolved) return;
    resolved = true;
    if (onReady) { onReady(state); onReady = null; }
  };

  const fallbackState = {
    user: { displayName: 'Гость отеля', photoURL: '', balance: 100, id: 'guest_123' },
    app: { shopId: 'demo_shop', platform: 'web' },
    colors: null
  };

  const setupNotibot = () => {
    const timeout = setTimeout(() => {
      _state = fallbackState;
      handleReady(_state);
    }, 400);

    window.notibot.onUpdate(function(user, app) {
      clearTimeout(timeout);
      _state = { 
        user: user || {}, 
        app: app || {}, 
        colors: app ? app.colors : null 
      };
      handleReady(_state);
      _listeners.forEach(fn => fn(_state));
    });
  };

  if (window.self !== window.top) {
    if (window.notibot) {
      setupNotibot();
    } else {
      // Загружаем скрипт локально только если мы во фрейме
      const script = document.createElement('script');
      script.src = './js/notibot-bridge.js';
      script.onload = () => {
        if (window.notibot) {
          setupNotibot();
        } else {
          _state = fallbackState;
          handleReady(_state);
        }
      };
      script.onerror = () => {
        _state = fallbackState;
        handleReady(_state);
      };
      document.head.appendChild(script);
    }
  } else {
    // Вне фрейма — мгновенный старт без загрузки скрипта
    _state = fallbackState;
    handleReady(_state);
  }
}

/** Подписаться на обновления (баланс, тема) */
export function onStateUpdate(fn) { _listeners.push(fn); }

/** Текущее состояние */
export function getState() { return _state; }

// Навигация
export function goToProduct(id) { 
  if (window.notibot) {
    id ? window.notibot.openProduct(id) : window.notibot.openStorefront(); 
  }
}

export function goToArticle(id) { 
  if (window.notibot) {
    id ? window.notibot.openArticle(id) : window.notibot.openStorefront(); 
  }
}

export function goToStorefront() { 
  if (window.notibot) window.notibot.openStorefront(); 
}

export function goToUserCard() { 
  if (window.notibot) window.notibot.openUserCard(); 
}

// Формы
export async function submitForm(formId, answers) {
  if (window.notibot) {
    return window.notibot.submitForm(formId, answers);
  }
  console.log('Mock submitForm:', formId, answers);
  return { success: true };
}

// Тема
function _applyTheme(colors) {
  if (!colors) return;
  const r = document.documentElement;
  r.style.setProperty('--color-bg', colors.background || '#090a11');
  r.style.setProperty('--color-text', colors.textPrimary || '#f8fafc');
  r.style.setProperty('--color-muted', colors.textSecondary || '#94a3b8');
  r.style.setProperty('--color-accent', colors.primaryMain || '#d4af37');
}
